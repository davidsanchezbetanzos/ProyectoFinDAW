Table usuario {
  id bigint [pk]
  nick varchar
  email varchar
  admin boolean
  equipo_id bigint [ref: > equipo.id]
  puntos bigint
}

Table equipo {
  id bigint [pk]
  nombre varchar
}

Table jornada {
  id bigint [pk]
  fechaini timestamp
  fechafin timestamp
  id_partido1 bigint [ref: > partido.id]
  id_partido2 bigint [ref: > partido.id]
  id_partido3 bigint [ref: > partido.id]
}

Table partido {
id bigint [pk]
jornada_id bigint [ref: > jornada.id]
equipolocal varchar
equipovisitante varchar
goleslocal bigint
golesvisitante bigint
quiniela varchar
}

Table pronostico {
id bigint [pk]
id_usuario bigint  [ref: > usuario.id]
id_jornada bigint  [ref: > jornada.id]
goleslocal_partido1 bigint
golesvisitante_partido1 bigint
goleslocal_partido2 bigint
golesvisitante_partido2 bigint
goleslocal_partido3 bigint
golesvisitante_partido3 bigint
quiniela_partido1 varchar
quiniela_partido2 varchar
quiniela_partido3 varchar
}
