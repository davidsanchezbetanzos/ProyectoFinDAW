Table usuario {
  id bigint [pk]
  nick varchar
  email varchar
  admin boolean
  equipo_id bigint [ref: > equipo.id]  
}

Table equipo {
  id bigint [pk]
  nombre varchar
}

Table jornada {
  id bigint [pk]
  fechaini timestamp
  fechafin timestamp
  estado varchar
 
}

Table partido {
id bigint [pk]
jornada_id bigint [ref: > jornada.id]
equipolocal varchar
equipovisitante varchar
goleslocal bigint
golesvisitante bigint
quiniela varchar
}

Table pronostico {
id bigint [pk]
id_usuario bigint  [ref: > usuario.id]
id_jornada bigint  [ref: > jornada.id]
}

Table pronostico_partido{
  id bigint [pk]
  id_pronostico bigint [ref: > pronostico.id]
  id_partido bigint [ref: > partido.id]
  goleslocal bigint
  golesvisitante bigint
  quiniela varchar
}
